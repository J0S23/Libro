name: render-book

# Reconstruye el libro con el código actual y actualiza la carpeta docs/ (la que sirve GitHub Pages).
on:
  push:
    branches: [main]
    paths-ignore:
      - "docs/**"
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: render-book
  cancel-in-progress: true

jobs:
  render:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: r-lib/actions/setup-pandoc@v2

      - uses: r-lib/actions/setup-r@v2
        with:
          use-public-rspm: true

      - name: Librerías del sistema
        run: |
          sudo apt-get update
          sudo apt-get install -y libcurl4-openssl-dev libssl-dev libxml2-dev \
            libfontconfig1-dev libharfbuzz-dev libfribidi-dev libfreetype6-dev \
            libpng-dev libtiff-dev libjpeg-dev libgit2-dev

      - name: Paquetes de R
        run: |
          Rscript -e 'install.packages(c("bookdown", "rmarkdown", "knitr", "tidyverse", "Amelia", "BSDA", "car", "coin", "effsize", "lmtest", "moments", "nortest", "openintro", "patchwork", "readxl", "rstatix", "sandwich"))'

      - name: Reconstruir el libro desde cero
        run: |
          rm -f docs/*.html docs/search_index.json docs/reference-keys.txt
          Rscript -e 'bookdown::render_book("Abstract.Rmd", "bookdown::gitbook")'
          FIRST=$(grep -oh 'data-path="[^"]*\.html"' docs/*.html | head -1 | cut -d'"' -f2)
          FIRST=${FIRST:-resumen.html}
          printf '<!DOCTYPE html>\n<html>\n<head>\n  <meta charset="UTF-8">\n  <meta http-equiv="refresh" content="0; url=%s">\n  <title>Libro</title>\n</head>\n<body>\n  <p>Redirigiendo al libro...</p>\n</body>\n</html>\n' "$FIRST" > docs/index.html
          touch docs/.nojekyll

      - name: Guardar docs/
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add -A docs
          if git diff --cached --quiet; then
            echo "Sin cambios en docs/"
          else
            git commit -m "Reconstruye el libro [skip ci]"
            git pull --rebase origin main
            git push
          fi
